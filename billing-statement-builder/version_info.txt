# PyInstaller version-file: embeds real metadata into the .exe (visible
# under right-click > Properties > Details on Windows) instead of shipping
# with no publisher/product info at all -- a blank-metadata unsigned exe
# is itself one of the signals SmartScreen/antivirus heuristics weigh, on
# top of it just being new/unsigned. This alone won't stop a SmartScreen
# warning (that's reputation- and signature-based, not metadata-based),
# but it's a real, honest description of what the file is and costs
# nothing to include.
VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(1, 0, 0, 0),
    prodvers=(1, 0, 0, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x4,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo(
      [
      StringTable(
        u'040904B0',
        [StringStruct(u'CompanyName', u'Coastal HPC'),
        StringStruct(u'FileDescription', u'Coastal HPC Billing Statement Builder'),
        StringStruct(u'FileVersion', u'1.0.0.0'),
        StringStruct(u'InternalName', u'CoastalHPCBillingStatementBuilder'),
        StringStruct(u'LegalCopyright', u'Coastal HPC'),
        StringStruct(u'OriginalFilename', u'Coastal HPC Billing Statement Builder.exe'),
        StringStruct(u'ProductName', u'Coastal HPC Billing Statement Builder'),
        StringStruct(u'ProductVersion', u'1.0.0.0')])
      ]),
    VarFileInfo([VarStruct(u'Translation', [1033, 1200])])
  ]
)
